cmake_minimum_required(VERSION 3.15)
cmake_policy(SET CMP0091 NEW)
project(${RIFE_NATIVE_LIBRARY} LANGUAGES C CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/patches.cmake")
set(CMAKE_CXX_STANDARD 11)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/settings.h"
    "${CMAKE_CURRENT_SOURCE_DIR}/layers.txt")
if(RIFE_STATIC_RUNTIME)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
else()
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
endif()
set(NCNN_BUILD_WITH_STATIC_CRT ${RIFE_STATIC_RUNTIME} CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
# The C ABI catches allocation/loader exceptions before they can reach Rust.
set(NCNN_DISABLE_EXCEPTION OFF CACHE BOOL "" FORCE)
if(MSVC)
    set(Vulkan_INCLUDE_DIR "${VULKAN_HEADERS}/include" CACHE PATH "" FORCE)
    set(Vulkan_LIBRARY "${CMAKE_BINARY_DIR}/vulkan-1.lib" CACHE FILEPATH "" FORCE)
    if(DEFINED VULKAN_MACHINE)
        # Set by Cargo for the target architecture, including Windows ARM64.
    elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
        set(VULKAN_MACHINE X64)
    else()
        set(VULKAN_MACHINE X86)
    endif()
    execute_process(COMMAND "${CMAKE_AR}" /nologo /def:${VULKAN_DEF} /machine:${VULKAN_MACHINE} /out:${Vulkan_LIBRARY} COMMAND_ERROR_IS_FATAL ANY)
    install(FILES "${Vulkan_LIBRARY}" DESTINATION lib)
endif()
foreach(flag NCNN_SHARED_LIB NCNN_OPENMP NCNN_STDIO NCNN_INT8 NCNN_BF16 NCNN_BUILD_BENCHMARK NCNN_BUILD_TESTS NCNN_BUILD_TOOLS NCNN_BUILD_EXAMPLES NCNN_PIXEL_ROTATE NCNN_PIXEL_AFFINE NCNN_PIXEL_DRAWING NCNN_C_API NCNN_INSTALL_SDK)
    set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(flag NCNN_VULKAN NCNN_THREADS NCNN_STRING NCNN_PIXEL NCNN_DISABLE_RTTI)
    set(${flag} ON CACHE BOOL "" FORCE)
endforeach()
file(STRINGS "${NCNN_SOURCE}/src/CMakeLists.txt" layer_lines REGEX "^ncnn_add_layer\\(")
foreach(line IN LISTS layer_lines)
    string(REGEX REPLACE "ncnn_add_layer\\(([A-Za-z0-9_]+).*" "\\1" name "${line}")
    string(TOLOWER "${name}" lower)
    set(WITH_LAYER_${lower} OFF CACHE BOOL "" FORCE)
endforeach()
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/layers.txt" enabled_layers)
set(RIFE_CUSTOM_LAYERS)
foreach(layer IN LISTS enabled_layers)
    if(layer MATCHES "^rife[.]")
        list(APPEND RIFE_CUSTOM_LAYERS "${layer}")
    else()
        string(TOLOWER "${layer}" layer)
        set(WITH_LAYER_${layer} ON CACHE BOOL "" FORCE)
    endif()
endforeach()
list(LENGTH RIFE_CUSTOM_LAYERS custom_layer_count)
if(NOT custom_layer_count EQUAL 1)
    message(FATAL_ERROR "Expected one RIFE custom layer in layers.txt")
endif()
list(GET RIFE_CUSTOM_LAYERS 0 RIFE_WARP_LAYER_NAME)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/native_layers.h"
    "#pragma once\n#define RIFE_WARP_LAYER_NAME \"${RIFE_WARP_LAYER_NAME}\"\n")
add_subdirectory("${NCNN_SOURCE}" ncnn)
# Keep NCNN diagnostics off the terminal, but retain the actual Vulkan/allocation
# failure on the invoking thread so the safe Rust API can report its cause.
set(platform_header "${CMAKE_CURRENT_BINARY_DIR}/ncnn/src/platform.h")
file(READ "${platform_header}" platform_source)
set(log_anchor "#define NCNN_LOGE(...)\n")
set(platform_checked "${platform_source}")
rife_replace_checked(platform_checked "logging" "${log_anchor}" "extern \"C\" void rife_record_native_error(const char*, ...) noexcept;\n#define NCNN_LOGE(...) rife_record_native_error(__VA_ARGS__)\n")
file(WRITE "${platform_header}" "${platform_checked}")

# The pinned NCNN version starts pipeline creation even after a failed weight
# read. Stop at that boundary so truncated models return an error, rather than
# dereferencing empty tensors. Generate an overlay; keep fetched sources intact.
file(READ "${NCNN_SOURCE}/src/net.cpp" net_source)
set(anchor "#if NCNN_VULKAN\n    if (opt.use_vulkan_compute)\n    {\n        if (!opt.pipeline_cache)")
set(checked_net "${net_source}")
rife_replace_checked(checked_net "model-load guard" "${anchor}" "    if (ret != 0) return ret; // Do not create pipelines from incomplete weights.\n\n${anchor}")
rife_replace_checked(checked_net "partial-model cleanup" "Layer* layer = d->layers[i];\n\n        Option opt1 = opt;" "Layer* layer = d->layers[i];\n        if (!layer) continue;\n\n        Option opt1 = opt;" 2)
# Cooperative checks at every pinned overload, avoiding unsafe termination.
string(PREPEND checked_net "#include \"cancellation.h\"\n")
foreach(method IN ITEMS upload_model forward_layer do_forward_layer)
    if(method STREQUAL "upload_model")
        set(expected_count 1)
    else()
        set(expected_count 3)
    endif()
    rife_regex_replace_checked(checked_net "${method} cancellation"
        "(int NetPrivate::${method}\\([^;{]*\n\\{)"
        "\\1\n    if (rife_cancelled()) return -233;" ${expected_count})
endforeach()
rife_replace_checked(checked_net "weight-load cancellation" "int lret = layer->load_model(mb);" "if (rife_cancelled()) return -233;\n        int lret = layer->load_model(mb);")
rife_replace_checked(checked_net "pipeline cancellation" "int cret = layer->create_pipeline(opt1);" "if (rife_cancelled()) return -233;\n        int cret = layer->create_pipeline(opt1);")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ncnn_net_checked.cpp" "${checked_net}")
get_target_property(ncnn_sources ncnn SOURCES)
list(REMOVE_ITEM ncnn_sources net.cpp "${NCNN_SOURCE}/src/net.cpp")
list(APPEND ncnn_sources "${CMAKE_CURRENT_BINARY_DIR}/ncnn_net_checked.cpp")
set_property(TARGET ncnn PROPERTY SOURCES "${ncnn_sources}")
target_sources(ncnn PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/cancellation.cpp")
target_include_directories(ncnn PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
# Backport Vulkan validity fixes without modifying the pinned NCNN checkout.
file(READ "${NCNN_SOURCE}/src/gpu.cpp" gpu_source)
set(matrix_anchor "std::vector<VkCooperativeMatrixPropertiesNV> properties(propertyCount);")
set(checked_gpu "${gpu_source}")
rife_replace_checked(checked_gpu "cooperative-matrix initialization" "${matrix_anchor}" "${matrix_anchor}\n            for (uint32_t i = 0; i < propertyCount; i++) { properties[i].sType = VK_STRUCTURE_TYPE_COOPERATIVE_MATRIX_PROPERTIES_NV; properties[i].pNext = 0; }")
set(storage_anchor "enabled16BitStorageFeatures.uniformAndStorageBuffer16BitAccess = VK_FALSE;")
set(storage_checked [=[enabled16BitStorageFeatures.uniformAndStorageBuffer16BitAccess = VK_FALSE;
    if (support_VK_KHR_get_physical_device_properties2 && info.support_VK_KHR_16bit_storage())
    {
        VkPhysicalDevice16BitStorageFeaturesKHR available16 = {};
        available16.sType = VK_STRUCTURE_TYPE_PHYSICAL_DEVICE_16BIT_STORAGE_FEATURES_KHR;
        VkPhysicalDeviceFeatures2KHR available = {};
        available.sType = VK_STRUCTURE_TYPE_PHYSICAL_DEVICE_FEATURES_2_KHR;
        available.pNext = &available16;
        vkGetPhysicalDeviceFeatures2KHR(info.physical_device(), &available);
        enabled16BitStorageFeatures.uniformAndStorageBuffer16BitAccess = available16.uniformAndStorageBuffer16BitAccess;
    }]=])
rife_replace_checked(checked_gpu "16-bit storage feature query" "${storage_anchor}" "${storage_checked}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ncnn_gpu_checked.cpp" "${checked_gpu}")
get_target_property(ncnn_sources ncnn SOURCES)
list(REMOVE_ITEM ncnn_sources gpu.cpp "${NCNN_SOURCE}/src/gpu.cpp")
list(APPEND ncnn_sources "${CMAKE_CURRENT_BINARY_DIR}/ncnn_gpu_checked.cpp")
set_property(TARGET ncnn PROPERTY SOURCES "${ncnn_sources}")
file(GLOB shaders CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/reference/*.comp")
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/settings.h" rgb_channel_setting REGEX "^#define RIFE_RGB_CHANNELS [0-9]+$")
list(LENGTH rgb_channel_setting rgb_channel_setting_count)
if(NOT rgb_channel_setting_count EQUAL 1)
    message(FATAL_ERROR "Expected one RIFE_RGB_CHANNELS value in settings.h")
endif()
list(GET rgb_channel_setting 0 rgb_channel_definition)
string(REGEX REPLACE "^#define RIFE_RGB_CHANNELS ([0-9]+)$" "\\1" RIFE_RGB_CHANNELS "${rgb_channel_definition}")
foreach(shader IN LISTS shaders)
    get_filename_component(name "${shader}" NAME_WE)
    file(READ "${shader}" text)
    string(FIND "${text}" "#version" start)
    string(SUBSTRING "${text}" ${start} -1 text)
    string(FIND "${text}" "\n" version_end)
    math(EXPR version_end "${version_end} + 1")
    string(SUBSTRING "${text}" 0 ${version_end} version)
    string(SUBSTRING "${text}" ${version_end} -1 shader_body)
    set(text "${version}#define RIFE_RGB_CHANNELS ${RIFE_RGB_CHANNELS}\n${shader_body}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${name}.text" "${text}")
    file(READ "${CMAKE_CURRENT_BINARY_DIR}/${name}.text" hex HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," hex "${hex}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${name}.comp.hex.h" "static const char ${name}_comp_data[] = {${hex}};\n")
endforeach()
add_library(${PROJECT_NAME} STATIC bridge.cpp reference/rife.cpp reference/warp.cpp)
target_include_directories(${PROJECT_NAME} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/reference")
target_link_libraries(${PROJECT_NAME} PUBLIC ncnn)
if(MSVC)
    target_compile_options(${PROJECT_NAME} PRIVATE /fp:fast)
    target_compile_definitions(${PROJECT_NAME} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
endif()
# Keep static-link order in one place; Cargo consumes the installed manifest.
set(native_libraries ${PROJECT_NAME} ncnn SPIRV glslang MachineIndependent OGLCompiler OSDependent GenericCodeGen)
install(TARGETS ${native_libraries} ARCHIVE DESTINATION lib)
string(REPLACE ";" "\n" link_manifest "${native_libraries}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/native-libraries.txt" "${link_manifest}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/native-libraries.txt" DESTINATION lib)
